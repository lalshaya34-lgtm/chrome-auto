const express = require('express'); const path = require('path'); const { chromium } = require('playwright'); const app = express(); const PORT = process.env.PORT || 3000; const state = { browser: null, page: null, timer: null, running: false }; async function ensureBrowser() { if (!state.browser) { state.browser = await chromium.launch({ headless: false, args: [ '--no-sandbox', '--disable-dev-shm-usage', '--disable-gpu' ] }); } if (!state.page) { state.page = await state.browser.newPage({ viewport: { width: 1366, height: 768 } }); } return state.page; } async function stopAutomation() { if (state.timer) { clearInterval(state.timer); state.timer = null; } state.running = false; } app.use(express.json({ limit: '1mb' })); app.use(express.static(path.join(__dirname, 'public'))); app.get('/api/health', (_req, res) => { res.json({ ok: true, running: state.running }); }); app.post('/api/open', async (req, res) => { try { const { url } = req.body || {}; if (!url) { return res.status(400).json({ error: 'URL is required.' }); } const page = await ensureBrowser(); await page.goto(url, { waitUntil: 'domcontentloaded', timeout: 60000 }); res.json({ ok: true, url }); } catch (error) { console.error('Open page error:', error); res.status(500).json({ error: error.message || 'Failed to open page.' }); } }); app.post('/api/start', async (req, res) => { try { const { mode = 'selector', selector = 'button', interval = 1000, x = 600, y = 350 } = req.body || {}; const page = await ensureBrowser(); if (state.running) { return res.json({ ok: true, running: true, message: 'Already running.' }); } state.running = true; state.timer = setInterval(async () => { try { if (mode === 'selector') { const locator = page.locator(selector).first(); const count = await locator.count(); if (count > 0) { await locator.click({ timeout: 3000 }).catch(() => {}); } return; } await page.mouse.click(Number(x), Number(y)); } catch (error) { console.error('Click loop error:', error); } }, Number(interval)); res.json({ ok: true, running: true }); } catch (error) { console.error('Start automation error:', error); res.status(500).json({ error: error.message || 'Failed to start automation.' }); } }); app.post('/api/stop', async (_req, res) => { try { await stopAutomation(); res.json({ ok: true, running: false }); } catch (error) { console.error('Stop automation error:', error); res.status(500).json({ error: error.message || 'Failed to stop automation.' }); } }); app.post('/api/close', async (_req, res) => { try { await stopAutomation(); if (state.browser) { await state.browser.close(); state.browser = null; state.page = null; } res.json({ ok: true }); } catch (error) { console.error('Close browser error:', error); res.status(500).json({ error: error.message || 'Failed to close browser.' }); } }); app.get('*', (_req, res) => { res.sendFile(path.join(__dirname, 'public', 'index.html')); }); app.listen(PORT, () => { console.log(`Chrome Auto Clicker app is running at http://localhost:${PORT}`); }); process.on('SIGINT', async () => { await stopAutomation(); if (state.browser) { await state.browser.close(); } process.exit(0); });
